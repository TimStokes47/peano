if(ENABLE_PARTICLES)
	message(STATUS "Configuring ToolboxParticles")

	file(GLOB_RECURSE TOOLBOX_PARTICLES_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)

	foreach(dimension IN LISTS PEANO_DIMENSIONS)
		foreach(mode IN LISTS BUILD_MODES)
			add_library(ToolboxParticles${dimension}d${SUFFIX_${mode}} STATIC)
			target_sources(ToolboxParticles${dimension}d${SUFFIX_${mode}} PRIVATE ${TOOLBOX_PARTICLES_SOURCES})
			target_include_directories(ToolboxParticles${dimension}d${SUFFIX_${mode}} PRIVATE ${CMAKE_SOURCE_DIR}/src)
			target_link_libraries(ToolboxParticles${dimension}d${SUFFIX_${mode}} PRIVATE Peano4Core${dimension}d${SUFFIX_${mode}}) 
			target_link_libraries(ToolboxParticles${dimension}d${SUFFIX_${mode}} PRIVATE Tarch${SUFFIX_${mode}}) 
			target_link_options(ToolboxParticles${dimension}d${SUFFIX_${mode}} PRIVATE ${${mode}_LDFLAGS})
			target_compile_options(ToolboxParticles${dimension}d${SUFFIX_${mode}} PRIVATE ${${mode}_FLAGS})
			target_compile_definitions(ToolboxParticles${dimension}d${SUFFIX_${mode}} PRIVATE ${${mode}_DEFINITIONS} Dimensions=${dimension})
			#target_precompile_headers(ToolboxParticles${dimension}d${SUFFIX_${mode}} PRIVATE ${CMAKE_SOURCE_DIR}/src/StdAfx.hpp)
		endforeach()
	endforeach()
endif()
